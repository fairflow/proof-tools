/-! ## The calculus, as the source states it (‹figure, page›)

One constructor for each primitive rule; each cites its rule.  Side

inductive Fam : Seq → ‹Type or Prop› where
  /-- ‹the rule's name in the source›, ‹figure, page or label›. -/
  | ‹rule› {‹schematic variables›} (‹side condition› : ‹…›) (‹premise› : Fam ‹…›) :
      Fam ‹variables and constructors only›
  -- as compiled:
  -- | andR {Γ : List Form} {A B : Form} (d₁ : Fam (Γ ⊢ A)) (d₂ : Fam (Γ ⊢ B)) : Fam (Γ ⊢ A ⋏ B)

/-- A control for `#slime`: no computed index. -/
inductive CleanControl : List Nat → Prop where
  | nil : CleanControl []
  | cons {n : Nat} {ns : List Nat} (h : CleanControl ns) : CleanControl (n :: ns)

/-- A control for `#slime`: a computed index in a conclusion. -/
inductive SlimedControl : List Nat → Prop where
  | nil : SlimedControl []
  | app {ms ns : List Nat} (h₁ : SlimedControl ms) (h₂ : SlimedControl ns) :
      SlimedControl (ms ++ ns)

/-! ## The proof system -/

/-- **A move**: a rule, named by its data. -/
inductive Move where
  | ‹rule› (‹what says which instance of the rule is meant›)
  deriving DecidableEq, Repr, Lean.ToExpr

/-- What a move asks of a judgment, what it leaves, and what to say when
it does not apply. -/
structure Rule where
  ok : Bool
  prems : List Seq
  why : String

/-- **The rule of a move at a judgment.** -/
def Move.rule (j : Seq) : Move → Rule
  | ‹.rule …› => ⟨‹the side conditions, as a Bool›, ‹the premises›,
      ‹"which side condition failed, in the source's words"›⟩
  -- as compiled:
  -- | .andL a b => ⟨decide (Form.and a b ∈ j.hyps), [⟨a :: b :: j.hyps, j.concl⟩],
  --     "the conjunction is not among the hypotheses"⟩
  -- | .andR =>
  --   match j.concl with
  --   | .and a b => ⟨true, [⟨j.hyps, a⟩, ⟨j.hyps, b⟩], ""⟩
  --   | _ => ⟨false, [], "the conclusion is not a conjunction"⟩

/-- **One step**: the premises of the move at the judgment, or why it does
not apply. -/
def step (m : Move) (j : Seq) : Except String (List Seq) :=
  if (m.rule j).ok then .ok (m.rule j).prems else .error (m.rule j).why

/-- **The calculus as a proof system.**  An `abbrev`, so that `S.Judg`
unfolds to `Seq` where Lean looks for instances. -/
abbrev S : ProofSystem where
  Judg := Seq
  Move := Move
  step := step

theorem step_of {m : Move} {j : Seq} (h : (m.rule j).ok = true) :
    S.step m j = .ok (m.rule j).prems := by
  show step m j = _
  simp [step, h]

theorem step_ok {m : Move} {j : Seq} {ps : List Seq} (h : S.step m j = .ok ps) :
    (m.rule j).ok = true ∧ ps = (m.rule j).prems := by
  have h' : step m j = .ok ps := h
  unfold step at h'
  split at h'
  · next hok => exact ⟨hok, (Except.ok.inj h').symm⟩
  · cases h'

/-! ### What holds: as the plan settled it.  Here, the family itself.
(The other choice is validity in a semantics, a proposition `Valid`: then
S1 is `theorem step_sound : S.Sound Valid`, as `sound` in the library's
`Calculus.lean`, and `famOfDer`, `Derivable` below do not arise.) -/

/-- **S1.  The soundness of the moves**: each move that applies is a rule of
`Fam`, applied to the derivations of its premises.  One case for each move;
a case where the move does not apply is not written (there `hok` is
`false = true`).  With `Fam` into `Prop`: `theorem`, and `have e'` for
`let e'`; `step_sound.sound : S.Sound Fam` is then its form by membership.
This is not the source's soundness theorem. -/
def step_sound : S.SoundT Fam := fun m j _ h e =>
  have hok : (m.rule j).ok = true := (step_ok h).1
  let e' : Each Fam (m.rule j).prems := (step_ok h).2 ▸ e
  match m, j, hok, e' with
  | ‹.rule …›, ‹the judgment, as far as the rule looks into it›, hok, e' =>
      ‹the constructor: its side conditions from `hok`, its premises `e'.1`, `e'.2.1`, …›
  -- as compiled:
  -- | .andL a b, ⟨_, _⟩, hok, e' => .andL a b (of_decide_eq_true hok) e'.1
  -- | .andR, ⟨_, .and _ _⟩, _, e' => .andR e'.1 e'.2.1

/-- **The map out**: a derivation of the proof system, as a derivation of
the calculus as the source states it (data, where `Fam` is into `Type`). -/
def famOfDer {j : Seq} (d : Der S j) : Fam j := Der.mapOut step_sound d

/-- That there is a derivation: the proposition that the engine, and a
certifier against what holds, take (part (b) §3). -/
def Derivable (j : Seq) : Prop := Nonempty (Fam j)

theorem step_sound_derivable : S.Sound Derivable := step_sound.nonempty

/-- **S2**, OPEN until proved: a named proposition, with no `sorry`. -/
def FamToDer : Prop := ∀ j : Seq, Fam j → Nonempty (Der S j)
